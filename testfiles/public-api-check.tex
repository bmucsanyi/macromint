\section{Theorems}

\begin{definition}
  A set is invariant when it maps into itself.
\end{definition}

\begin{theorem}
  Every fixed point is a point.
\end{theorem}

\begin{proposition}
  A linear map preserves linear combinations.
\end{proposition}

\begin{lemma}
  If $x = y$, then $y = x$.
\end{lemma}
